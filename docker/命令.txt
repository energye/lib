系统启动
sudo systemctl enable docker

拉取镜像
docker pull

查看镜像
docker images 

拉取 debian 11 版本镜像
docker pull --platform i386 debian:bullseye
docker pull --platform amd64 debian:bullseye

docker pull --platform=linux/arm   arm32v7/debian:bullseye
docker pull --platform=linux/arm64 arm64v8/debian:bullseye

拉取 ubuntu 18.04 版本镜像

docker pull --platform=linux/i386  i386/ubuntu:18.04
docker pull --platform=linux/amd64 amd64/ubuntu:18.04

docker pull --platform=linux/arm   arm32v7/ubuntu:18.04
docker pull --platform=linux/arm64 arm64v8/ubuntu:18.04
